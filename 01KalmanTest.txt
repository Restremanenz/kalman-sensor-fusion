import os
import json
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.spatial.transform import Rotation as R
from filterpy.kalman import KalmanFilter

from st_log_reader import STLogReader

class IMUCalibration:
    """Lädt die Kalibrierungsparameter und wendet sie an."""
    def __init__(self, acc_json_path, gyro_json_path):
        with open(acc_json_path, 'r') as f:
            acc_data = json.load(f)
            self.acc_bias = np.array(acc_data["offset_b"])
            self.acc_M = np.array(acc_data["matrix_M"])
            
        with open(gyro_json_path, 'r') as f:
            gyro_data = json.load(f)
            self.gyro_bias = np.array(gyro_data["offset_b"])
            self.gyro_noise_std = np.array(gyro_data["noise_std"]) * (np.pi / 180.0)

    def calibrate_acc(self, raw_acc):
        calibrated = self.acc_M @ (raw_acc - self.acc_bias)
        return calibrated * 9.81

    def calibrate_gyro(self, raw_gyro):
        calibrated_dps = raw_gyro - self.gyro_bias
        return calibrated_dps * (np.pi / 180.0)

class FilterpyESKF:
    """
    Error-State Kalman Filter basierend auf filterpy.
    Nominaler Zustand: Position (3), Velocity (3), Quaternion (4)
    Error Zustand (im Filter): d_Pos (3), d_Vel (3), d_Theta (3) = 9 Dimensionen
    """
    def __init__(self, initial_pos, gyro_noise_std):
        # --- 1. Nominaler Zustand (Die echte Physik) ---
        self.p = np.array(initial_pos, dtype=float)
        self.v = np.zeros(3)
        self.q = R.from_quat([0, 0, 0, 1])
        self.g = np.array([0, 0, -9.81]) 

        # --- 2. Filterpy Setup für den Error-State ---
        self.kf = KalmanFilter(dim_x=9, dim_z=3) # x = 9D Fehler, z = 3D Messung
        self.kf.x = np.zeros(9) # Initialer Fehler ist 0
        self.kf.P = np.eye(9) * 0.01 # Initiale Unsicherheit
        
        # Prozessrauschen (Q) Basis-Matrix (6x6: 3D Accel, 3D Gyro)
        self.base_Q = np.eye(6)
        self.base_Q[0:3, 0:3] *= (0.1)**2  # TUNING: Accel Noise (m/s^2)^2
        self.base_Q[3:6, 3:6] = np.diag(gyro_noise_std**2) # Gyro Noise

        # Messrauschen für die verschiedenen Updates (R)
        self.R_grav = np.eye(3) * (0.5)**2  # TUNING: Wie stark trauen wir der Wasserwaage?
        self.R_zupt = np.eye(3) * (0.05)**2 # TUNING: Wie sicher sind wir beim Stillstand? (m/s)

    def predict(self, acc, gyro, dt):
        """Koppelnavigation und Fehlerpropagierung."""
        
        # --- Nominal State Update (Physik) ---
        acc_world = self.q.apply(acc)
        
        self.p += self.v * dt + 0.5 * (acc_world + self.g) * dt**2
        self.v += (acc_world + self.g) * dt
        self.q = self.q * R.from_rotvec(gyro * dt)

        # --- Error State Update (Filterpy) ---
        # State Transition Matrix F (Jacobian)
        F_x = np.eye(9)
        F_x[0:3, 3:6] = np.eye(3) * dt
        
        acc_skew = np.array([
            [0, -acc_world[2], acc_world[1]],
            [acc_world[2], 0, -acc_world[0]],
            [-acc_world[1], acc_world[0], 0]
        ])
        F_x[3:6, 6:9] = -acc_skew * dt

        # Noise Mapping Matrix
        F_i = np.zeros((9, 6))
        F_i[3:6, 0:3] = np.eye(3) * dt
        F_i[6:9, 3:6] = np.eye(3) * dt

        # Dynamisches Zuweisen von F und Q in filterpy
        self.kf.F = F_x
        self.kf.Q = F_i @ self.base_Q @ F_i.T
        
        # Prädiktion im Filter durchführen (Propagiert im Wesentlichen nur P, da x=0)
        self.kf.predict()

    def _inject_error_and_reset(self):
        """Injiziert den berechneten Fehler in die Physik und resetet den Filter."""
        # Update Physik mit berechnetem Fehler
        self.p += self.kf.x[0:3]
        self.v += self.kf.x[3:6]
        
        error_rot = R.from_rotvec(self.kf.x[6:9])
        self.q = error_rot * self.q
        
        # SEHR WICHTIG FÜR ESKF: Fehler zurücksetzen!
        self.kf.x = np.zeros(9)

    def update_zupt(self):
        """Zero Velocity Update: Zwingt Geschwindigkeit auf 0 und repariert Position/Lage."""
        H = np.zeros((3, 9))
        H[0:3, 3:6] = np.eye(3) # Wir messen die Velocity (Index 3,4,5)
        
        # Innovation: Gemessen (0,0,0) - Vorhergesagt (self.v)
        innovation = np.array([0.0, 0.0, 0.0]) - self.v
        
        # Filterpy Update (z wird hier direkt als Innovation übergeben, da H*x = 0)
        self.kf.update(z=innovation, R=self.R_zupt, H=H)
        self._inject_error_and_reset()

    def update_gravity(self, acc):
        """Wasserwaagen-Update bei Stillstand gegen Roll/Pitch Drift."""
        acc_norm = np.linalg.norm(acc)
        if abs(acc_norm - 9.81) > 0.5:
            return 

        a_dir = acc / acc_norm
        g_pred = self.q.inv().apply([0, 0, 1])

        H = np.zeros((3, 9))
        H[0:3, 6:9] = np.array([
            [0, -g_pred[2], g_pred[1]],
            [g_pred[2], 0, -g_pred[0]],
            [-g_pred[1], g_pred[0], 0]
        ])

        innovation = a_dir - g_pred
        self.kf.update(z=innovation, R=self.R_grav, H=H)
        self._inject_error_and_reset()


def main():
    log_folder = "./Data/log6" 
    reader = STLogReader(log_folder)
    calib = IMUCalibration("acc_param.json", "gyro_bias.json")
    
    df_acc = reader.get_sensor_data("lsm6dsv16x_acc")
    df_gyro = reader.get_sensor_data("lsm6dsv16x_gyro")

    df_acc = df_acc.sort_values("Time")
    df_gyro = df_gyro.sort_values("Time")
    df_imu = pd.merge_asof(df_acc, df_gyro, on="Time", direction="nearest")
    
    eskf = FilterpyESKF(initial_pos=[0.0, 0.0, 0.0], gyro_noise_std=calib.gyro_noise_std)
    
    positions = []
    times = df_imu["Time"].values
    
    # Tuning Parameter für ZUPT
    ZUPT_THRESHOLD = 0.5 # m/s^2 (Grenz-Wert für lineare Beschleunigung)
    
    for i in range(1, len(df_imu)):
        dt = times[i] - times[i-1]
        if dt <= 0: continue
            
        row = df_imu.iloc[i]
        
        # Spaltennamen ggf. an deine echten Namen anpassen!
        raw_acc = np.array([row['A_x [g]'], row['A_y [g]'], row['A_z [g]']]) 
        raw_gyro = np.array([row['G_x [dps]'], row['G_y [dps]'], row['G_z [dps]']]) 
        
        acc_calib = calib.calibrate_acc(raw_acc)
        gyro_calib = calib.calibrate_gyro(raw_gyro)
        
        # 1. Prediction (IMU vorwärts rechnen)
        eskf.predict(acc_calib, gyro_calib, dt)
        
        # 2. Heuristik berechnen (Wie stark bewegen wir uns?)
        # Lineare Beschleunigung berechnen (Gravitation abziehen)
        acc_world = eskf.q.apply(acc_calib)
        linear_acc = acc_world + eskf.g
        acc_magnitude = np.linalg.norm(linear_acc)
        
        # 3. ZUPT und Gravity Update anwenden, wenn in Ruhe!
        if acc_magnitude < ZUPT_THRESHOLD:
            eskf.update_zupt()
            eskf.update_gravity(acc_calib)
            
        positions.append(eskf.p.copy())
        
    # --- Plotten ---
    positions = np.array(positions)
    
    fig = plt.figure(figsize=(10, 8))
    ax = fig.add_subplot(111, projection='3d')
    ax.plot(positions[:, 0], positions[:, 1], positions[:, 2], label='ESKF mit ZUPT', color='b')
    ax.scatter(positions[0, 0], positions[0, 1], positions[0, 2], color='g', s=100, label='Start')
    ax.scatter(positions[-1, 0], positions[-1, 1], positions[-1, 2], color='r', s=100, label='Ende')
    ax.set_xlabel('X [m]'), ax.set_ylabel('Y [m]'), ax.set_zlabel('Z [m]')
    ax.legend()
    plt.show()

if __name__ == "__main__":
    main()
